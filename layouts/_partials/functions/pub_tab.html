{{/* Map a page to its Writings tab id. Input: page. Output: string */}} {{- $types := .Params.publication_types | default slice -}} {{- $tab := "articles" -}} {{- $legacy := index site.Data "writings_legacy_map" -}} {{- $slug := path.Base (path.Dir .File.Path) -}} {{- if and $legacy (index $legacy.entries $slug) }}{{ $tab = (index $legacy.entries $slug).tab }} {{- else if eq .Section "talk" }}{{ $tab = "presentations" }} {{- else if eq .Section "software" }}{{ $tab = "software" }} {{- else if in $types "journal_article" }}{{ $tab = "articles" }} {{- else if or (in $types "book") (in $types "book_chapter") (in $types "chapter") }}{{ $tab = "chapters" }} {{- else if in $types "data" }}{{ $tab = "data" }} {{- else if in $types "presentation" }}{{ $tab = "presentations" }} {{- else if in $types "software" }}{{ $tab = "software" }} {{- else if eq .Params.status "in_progress" }}{{ $tab = "in-progress" }} {{- else if or (in $types "report") (in $types "working_paper") }}{{ $tab = "working-papers" }} {{- end -}} {{- return $tab -}}
